Koa Ola Ranch Relational Schema

Underlined attributes are primary keys. Foreign keys are listed below each table.

Animal(AnimalID, Name, Breed, DateOfBirth, Gender, Weight, AnimalType)

HealthRecord(HealthRecordID, AnimalID, CheckupDate, MedicineGiven, Diagnosis, Treatment)

ForeignKey AnimalID reference Animal

Programs(ProgramID, ProgramName, ProgramDate, Description)

Program_Class(ClassID, ProgramID, ClassName, ClassDate)

ForeignKey ProgramID reference Programs

Reservation(ReservationID, GuestID, ClassID, ReservationDate)

ForeignKey GuestID reference Guest

ForeignKey ClassID reference Program_Class

Patrons(PatronID, PatronName, Email, Phone)

Donations(DonationID, PatronID, ProgramID, Amount, DonationDate)

ForeignKey PatronID reference Patrons

ForeignKey ProgramID reference Programs

Grants(GrantID, ProgramID, GrantName, FundingSource, Amount, DateReceived)

ForeignKey ProgramID reference Programs

Guest(GuestID, FirstName, LastName, DateOfBirth, Phone)

Training(TrainingID, TrainingLocation, TrainingName, TrainingDate)

AssociateTraining(AssociateID, TrainingID, Description)

ForeignKey AssociateID reference Associate

ForeignKey TrainingID reference Training

Associate(AssociateID, FirstName, LastName, HireDate, RetireDate, TerminationDate, Benefits)

Promotion(PromotionID, AssociateID, PromotionDate)

ForeignKey AssociateID reference Associate

SocialMedia(PostID, AssociateID, Platform, PostDate)

ForeignKey AssociateID reference Associate
